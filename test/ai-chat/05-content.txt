<?php namespace Elv\Site\Classes;

use Tailor\Models\EntryRecord;
use Tailor\Models\StructureRecord;

/**
 * Content — выборка Tailor-записей для страниц темы.
 *
 * Запросы держатся в одном месте: шаблоны не знают про Tailor-модели,
 * а разделы (section) резолвятся по коду, а не по числовому id.
 */
class Content
{
    /**
     * @var array<string, array>
     */
    protected static $cache = [];

    /**
     * flush сбрасывает кеш выборок.
     *
     * Живому запросу он не нужен (кеш умирает вместе с запросом), но тесты
     * и импорт выполняются в одном процессе: без сброса следующий тест увидит
     * записи предыдущего, а карта сайта — удалённую запись.
     */
    public static function flush(): void
    {
        static::$cache = [];
    }

    /**
     * records возвращает опубликованные записи раздела по сортировке.
     *
     * @return array<int, EntryRecord>
     */
    public static function records(string $handle, $limit = null): array
    {
        $key = 'records|' . $handle . '|' . (string) $limit;
        if (array_key_exists($key, static::$cache)) {
            return static::$cache[$key];
        }

        $query = EntryRecord::inSection($handle)
            ->applyVisibleFrontend()
            ->orderBy('sort')
            ->orderBy('id');

        if ($limit !== null) {
            $query->limit((int) $limit);
        }

        return static::$cache[$key] = $query->get()->all();
    }

    /**
     * recordsInSection возвращает записи одного раздела по его коду.
     *
     * @return array<int, EntryRecord>
     */
    public static function recordsInSection(string $handle, string $sectionHandle, string $code, $limit = null): array
    {
        $id = static::sectionId($sectionHandle, $code);
        if ($id === null) {
            return [];
        }

        $key = 'records-section|' . $handle . '|' . $id . '|' . (string) $limit;
        if (array_key_exists($key, static::$cache)) {
            return static::$cache[$key];
        }

        $query = EntryRecord::inSection($handle)
            ->applyVisibleFrontend()
            ->where('section_id', $id)
            ->orderBy('sort')
            ->orderBy('id');

        if ($limit !== null) {
            $query->limit((int) $limit);
        }

        return static::$cache[$key] = $query->get()->all();
    }

    /**
     * record находит запись по slug.
     */
    public static function record(string $handle, $slug): ?EntryRecord
    {
        $slug = trim((string) $slug);
        if ($slug === '') {
            return null;
        }

        $key = 'record|' . $handle . '|' . $slug;
        if (array_key_exists($key, static::$cache)) {
            return static::$cache[$key];
        }

        $record = EntryRecord::inSection($handle)
            ->applyVisibleFrontend()
            ->where('slug', $slug)
            ->first();

        return static::$cache[$key] = $record;
    }

    /**
     * sections возвращает разделы инфоблока по сортировке.
     *
     * @return array<int, StructureRecord>
     */
    public static function sections(string $handle): array
    {
        $key = 'sections|' . $handle;
        if (array_key_exists($key, static::$cache)) {
            return static::$cache[$key];
        }

        return static::$cache[$key] = StructureRecord::inSection($handle)
            ->where('is_enabled', true)
            ->orderBy('sort')
            ->orderBy('id')
            ->get()
            ->all();
    }

    /**
     * sectionId возвращает id раздела по его коду.
     */
    public static function sectionId(string $handle, string $code): ?int
    {
        foreach (static::sections($handle) as $section) {
            if ((string) $section->code === (string) $code) {
                return (int) $section->id;
            }
        }

        return null;
    }

    /**
     * sectionCode возвращает код раздела по его id.
     */
    public static function sectionCode(string $handle, $id): string
    {
        foreach (static::sections($handle) as $section) {
            if ((int) $section->id === (int) $id) {
                return (string) $section->code;
            }
        }

        return '';
    }

    /**
     * sectionTitle возвращает название раздела по его id.
     */
    public static function sectionTitle(string $handle, $id): string
    {
        foreach (static::sections($handle) as $section) {
            if ((int) $section->id === (int) $id) {
                return (string) $section->title;
            }
        }

        return '';
    }
}

<?php namespace Elv\Site\Classes;

use Str;
use MediaLibrary;
use Tailor\Models\StructureRecord;

/**
 * PageBuilder — страница, собранная из блоков Tailor.
 *
 * Модель данных — blueprint `Elv\Page` (type `structure`): у записи есть
 * заголовок, адрес и повторитель `blocks` с группами из `Elv\Blocks`.
 * Здесь только чтение и подготовка данных к рендеру: шаблоны не знают ни про
 * Tailor-модели, ни про то, как хранятся группы повторителя.
 *
 * Три правила, которые держат конструктор в рамках проекта:
 *
 * 1. Белый список блоков и разделов: ключи групп совпадают с именами
 *    партиалов `themes/elv/partials/blocks/*.htm`, ключи разделов — с
 *    `SOURCES`. Неизвестная группа пропускается, а не роняет страницу.
 * 2. Отключённая страница не отдаётся: выборка идёт через
 *    `applyVisibleFrontend()` — тот же фильтр, что у новостей и книг.
 * 3. Блок «сетка записей» берёт записи только через `Content::records()`:
 *    второй источник выборки означал бы второй фильтр опубликованности.
 */
class PageBuilder
{
    /**
     * @var string SECTION blueprint-раздел страниц
     */
    public const SECTION = 'Elv\\Page';

    /**
     * @var int MAX_LIMIT потолок записей в блоке «сетка»
     */
    public const MAX_LIMIT = 12;

    /**
     * @var array<int, string> GROUPS группы повторителя, они же имена партиалов
     */
    public const GROUPS = ['text', 'text-image', 'quote', 'records', 'video', 'toc'];

    /**
     * @var array<string, array<string, string>> SOURCES разделы для блока «сетка»
     */
    public const SOURCES = [
        'books' => [
            'handle' => 'Elv\\Book',
            'card' => 'cards/book',
            'title' => 'Сувениры и подарки',
            'url' => '/books/',
        ],
        'collections' => [
            'handle' => 'Elv\\Collection',
            'card' => 'cards/collection',
            'title' => 'Подборки',
            'url' => '/collections/',
        ],
        'events' => [
            'handle' => 'Elv\\Event',
            'card' => 'cards/event',
            'title' => 'События',
            'url' => '/events/',
        ],
        'news' => [
            'handle' => 'Elv\\News',
            'card' => 'cards/news',
            'title' => 'Новости',
            'url' => '/news/',
        ],
        'printing' => [
            'handle' => 'Elv\\Printing',
            'card' => 'cards/print',
            'title' => 'Печать',
            'url' => '/printing/',
        ],
        'services' => [
            'handle' => 'Elv\\Service',
            'card' => 'cards/service',
            'title' => 'Услуги',
            'url' => '/services/',
        ],
    ];

    /**
     * @var array<string, array<int, string>> FIELDS поля каждой группы
     */
    public const FIELDS = [
        'text' => ['title', 'body'],
        'text-image' => ['title', 'body', 'image', 'image_position', 'caption'],
        'quote' => ['kind', 'text', 'author'],
        'records' => ['title', 'source', 'limit'],
        'video' => ['title', 'file', 'poster', 'caption'],
        'toc' => ['title', 'include_h3'],
    ];

    /**
     * @var array<string, array|null> cache выборок по адресу
     */
    protected static $cache = [];

    /**
     * @var array<int, array{slug: string, lastmod: string}>|null cache для карты сайта
     */
    protected static $pages;

    /**
     * flush сбрасывает кеш выборок между тестами и импортом.
     */
    public static function flush(): void
    {
        static::$cache = [];
        static::$pages = null;
    }

    /**
     * pages отдаёт опубликованные страницы конструктора для карты сайта.
     *
     * Отдельный список нужен потому, что статические страницы темы перечислены
     * в `Sitemap::PAGES` руками, а собранные в админке появляются и исчезают
     * без правки кода — карта сайта обязана видеть их через БД.
     *
     * @return array<int, array{slug: string, lastmod: string}>
     */
    public static function pages(): array
    {
        if (static::$pages !== null) {
            return static::$pages;
        }

        $pages = [];

        $records = StructureRecord::inSection(static::SECTION)
            ->applyVisibleFrontend()
            ->whereNull('parent_id')
            ->get();

        foreach ($records as $page) {
            $slug = trim((string) $page->slug);

            if ($slug === '') {
                continue;
            }

            $pages[] = [
                'slug' => $slug,
                'lastmod' => date('Y-m-d', strtotime((string) $page->updated_at ?: 'now')),
            ];
        }

        return static::$pages = $pages;
    }

    /**
     * find находит опубликованную страницу по адресу.
     *
     * Берутся только страницы верхнего уровня: адрес страницы — `/slug/`,
     * вложенность в конструкторе не поддерживается (`structure.maxDepth: 1`),
     * иначе `fullslug` перестал бы совпадать с адресом.
     */
    public static function find(string $slug): ?StructureRecord
    {
        $slug = trim($slug);

        if ($slug === '') {
            return null;
        }

        if (array_key_exists($slug, static::$cache)) {
            return static::$cache[$slug];
        }

        $page = StructureRecord::inSection(static::SECTION)
            ->applyVisibleFrontend()
            ->where('slug', $slug)
            ->whereNull('parent_id')
            ->first();

        return static::$cache[$slug] = $page;
    }

    /**
     * prepare готовит блоки к рендеру.
     *
     * Возвращает `rows` (по строке на блок: `group`, `id`, `data`) и `toc`
     * (оглавление страницы, собранное из подзаголовков всех текстовых блоков).
     *
     * @return array{rows: array<int, array>, toc: array<int, array>}
     */
    public static function prepare(?StructureRecord $page): array
    {
        $rows = [];
        $toc = [];
        $used = [];

        if (!$page) {
            return ['rows' => [], 'toc' => []];
        }

        foreach ($page->blocks as $item) {
            $group = (string) $item->content_group;

            if (!in_array($group, static::GROUPS, true)) {
                continue;
            }

            $data = static::data($group, $item);
            $data = static::applyHeadings($group, $data, $toc, $used);

            if ($group === 'records') {
                $data = static::applyRecords($data);
            }

            if ($group === 'quote') {
                $data['kind_class'] = ($data['kind'] ?? '') === 'warning' ? 'alert-warning' : 'alert-info';
            }

            if ($group === 'video' && isset($data['file'])) {
                $data['src'] = static::fileUrl((string) $data['file']);
            }

            $rows[] = [
                'group' => $group,
                'id' => (int) $item->id,
                'data' => $data,
            ];
        }

        return ['rows' => $rows, 'toc' => $toc];
    }

    /**
     * grid возвращает записи раздела для блока «сетка».
     *
     * Раздел вне белого списка даёт пустой массив, а не выборку «по всему»:
     * опечатка в YAML не должна вываливать на страницу весь инфоблок.
     *
     * @return array<int, \Tailor\Models\EntryRecord>
     */
    public static function grid(string $source, int $limit = 3): array
    {
        $meta = static::SOURCES[$source] ?? null;

        if ($meta === null) {
            return [];
        }

        $limit = max(1, min(static::MAX_LIMIT, $limit));

        return Content::records($meta['handle'], $limit);
    }

    /**
     * fileUrl превращает значение mediafinder в адрес файла.
     *
     * Внешние адреса отбрасываются: CSP разрешает медиа только со своего
     * домена (`media-src` не выводится, работает `default-src 'self'`),
     * поэтому вставленный чужой плеер всё равно не заработал бы.
     */
    public static function fileUrl(string $value): string
    {
        $path = Media::imagePath($value);

        if ($path === '' || preg_match('#^(https?:)?//#i', $path)) {
            return '';
        }

        if ($path[0] === '/') {
            return $path;
        }

        if (str_starts_with($path, 'storage/')) {
            return '/' . $path;
        }

        if (str_starts_with($path, 'assets/')) {
            return Media::themeUrl($path);
        }

        return (string) MediaLibrary::url($path);
    }

    /**
     * data собирает поля группы в массив для партиала.
     */
    protected static function data(string $group, $item): array
    {
        $data = [];

        foreach (static::FIELDS[$group] as $field) {
            $value = $item->{$field};
            $data[$field] = is_string($value) ? trim($value) : $value;
        }

        return $data;
    }

    /**
     * applyHeadings добавляет якоря подзаголовкам и собирает оглавление.
     *
     * Заголовок самого блока и `<h2>/<h3>` внутри richeditor нумеруются
     * в одном пространстве: два блока с одинаковым подзаголовком получают
     * `-2`, `-3`, иначе ссылка из оглавления ведёт не туда.
     */
    protected static function applyHeadings(string $group, array $data, array &$toc, array &$used): array
    {
        if (!in_array($group, ['text', 'text-image'], true)) {
            return $data;
        }

        $title = (string) ($data['title'] ?? '');

        if ($title !== '') {
            $data['title_id'] = static::anchor($title, $used);
            $toc[] = ['level' => 2, 'id' => $data['title_id'], 'text' => $title];
        }

        $data['body'] = static::bodyAnchors((string) ($data['body'] ?? ''), $toc, $used);

        return $data;
    }

    /**
     * bodyAnchors расставляет id у `<h2>/<h3>` и дополняет оглавление.
     */
    protected static function bodyAnchors(string $html, array &$toc, array &$used): string
    {
        if ($html === '' || !preg_match('/<h[23][\s>]/i', $html)) {
            return $html;
        }

        $result = preg_replace_callback('/<h([23])([^>]*)>(.*?)<\/h\1>/is', function ($matches) use (&$toc, &$used) {
            $level = (int) $matches[1];
            $attrs = (string) $matches[2];
            $text = trim(strip_tags((string) $matches[3]));

            if ($text === '') {
                return $matches[0];
            }

            if (preg_match('/\bid=["\']([^"\']+)["\']/i', $attrs, $found)) {
                $id = $found[1];
            }
            else {
                $id = static::anchor($text, $used);
                $attrs .= ' id="' . e($id) . '"';
            }

            $toc[] = ['level' => $level, 'id' => $id, 'text' => $text];

            return '<h' . $level . $attrs . '>' . $matches[3] . '</h' . $level . '>';
        }, $html);

        return (string) $result;
    }

    /**
     * anchor возвращает уникальный якорь из текста подзаголовка.
     */
    protected static function anchor(string $text, array &$used): string
    {
        $base = Str::slug($text);

        if ($base === '') {
            $base = 'section';
        }

        $id = $base;
        $counter = 2;

        while (isset($used[$id])) {
            $id = $base . '-' . $counter;
            $counter++;
        }

        $used[$id] = true;

        return $id;
    }

    /**
     * applyRecords дополняет блок «сетка» записями и адресами карточки.
     */
    protected static function applyRecords(array $data): array
    {
        $source = (string) ($data['source'] ?? '');
        $limit = (int) ($data['limit'] ?? 3);

        $data['items'] = static::grid($source, $limit);
        $data['card'] = static::SOURCES[$source]['card'] ?? '';
        $data['more_url'] = static::SOURCES[$source]['url'] ?? '';

        return $data;
    }
}

<?php namespace Elv\Site\Classes;

use Illuminate\Http\Request;

/**
 * FormFields — разбор полей формы в формате источника (Bitrix).
 *
 * Источник присылает вложенные имена (`PROPERTY[EMAIL][0]`), Laravel раскладывает
 * их в массив `PROPERTY => [EMAIL => [0 => …]]`. Здесь это превращается в плоский
 * набор `EMAIL`, `PHONE`, `DATE`, … — с ним работают валидация и модели, а не с
 * именами Битрикса.
 */
class FormFields
{
    /**
     * fromRequest собирает плоские поля из запроса.
     *
     * @return array<string, string>
     */
    public static function fromRequest(Request $request): array
    {
        return static::normalize((array) $request->all());
    }

    /**
     * normalize превращает `PROPERTY[*][0]` и простые имена в плоский массив.
     *
     * @param  array<string, mixed>  $input
     * @return array<string, string>
     */
    public static function normalize(array $input): array
    {
        $fields = [];

        foreach ((array) ($input['PROPERTY'] ?? []) as $name => $value) {
            $fields[static::key($name)] = static::scalar($value);
        }

        foreach ($input as $name => $value) {
            if (!is_string($name) || is_array($value)) {
                continue;
            }

            $key = static::key($name);
            if ($key !== '' && !array_key_exists($key, $fields)) {
                $fields[$key] = static::scalar($value);
            }
        }

        return array_filter($fields, static function ($value) {
            return $value !== '';
        });
    }

    /**
     * consentGiven отвечает, отмечен ли чекбокс согласия.
     *
     * @param  array<string, mixed>  $input
     */
    public static function consentGiven(array $input): bool
    {
        $fields = static::normalize($input);

    /**
     * minifyCss сжимает CSS, не трогая строки, url() и calc().
     */
    public static function minifyCss(string $css): string
    {
        $out = '';
        $length = strlen($css);
        $quote = '';
        $paren = 0;
        $space = false;

        for ($i = 0; $i < $length; $i++) {
            $char = $css[$i];

            if ($quote !== '') {
                $out .= $char;
                if ($char === '\\' && $i + 1 < $length) {
                    $out .= $css[++$i];
                    continue;
                }
                if ($char === $quote) {
                    $quote = '';
                }
                continue;
            }

            if ($char === '/' && $i + 1 < $length && $css[$i + 1] === '*') {
                $end = strpos($css, '*/', $i + 2);
                $i = $end === false ? $length : $end + 1;
                $space = true;
                continue;
            }

            if ($char === '"' || $char === "'") {
                if ($space && $out !== '' && !in_array(substr($out, -1), ['{', '}', ';', ','], true)) {
                    $out .= ' ';
                }
                $out .= $char;
                $quote = $char;
                $space = false;
                continue;
            }

            if ($char === '(') {
                $paren++;
            } elseif ($char === ')') {
                $paren = max(0, $paren - 1);
            }

            if (ctype_space($char)) {
                $space = true;
                continue;
            }

            if ($space && $out !== '') {
                $previous = substr($out, -1);
                $drop = in_array($previous, ['{', '}', ';', ','], true)
                    || ($paren === 0 && in_array($char, ['{', '}', ';', ','], true));
                if (!$drop) {
                    $out .= ' ';
                }
            }

            $out .= $char;
            $space = false;
        }

        return $out;
    }

<?php namespace Elv\Site\Classes;

/**
 * Rss — лента новостей `/rss.xml` в формате RSS 2.0.
 *
 * Источник отдавал `/news/rss/`; адрес сохранён смыслом, но не путём: лента
 * одна на сайт, поэтому `/rss.xml` — как у большинства статических сайтов.
 * Записи берутся из того же `Content::records()`, что и страница новостей,
 * поэтому отключённая или удалённая новость в ленту не попадает.
 *
 * Сортировка — по дате публикации (свежие сверху), а не по `sort` из админки:
 * лента читается подрядчиком, порядок в списке новостей для неё неважен.
 * Дата без `published_at` откатывается на `created_at`, иначе такая запись
 * уехала бы в конец ленты навсегда.
 */
class Rss
{
    /**
     * HANDLE — Tailor-раздел, из которого собирается лента.
     */
    public const HANDLE = 'Elv\News';

    /**
     * PREFIX — адрес списка новостей.
     */
    public const PREFIX = '/news/';

    /**
     * LIMIT — сколько записей отдавать: RSS-читалки не любят простыни.
     */
    public const LIMIT = 20;

    /**
     * xml собирает готовый документ ленты.
     *
     * Параметры с умолчаниями отдают новости (`/rss.xml`); блог вызывает
     * тот же код со своим разделом и адресом (`/blog/rss.xml`), чтобы формат
     * ленты не расползался на две копии.
     */
    public static function xml(
        string $baseUrl,
        string $handle = self::HANDLE,
        string $prefix = self::PREFIX,
        string $feedPath = '/rss.xml',
        string $channelTitle = '',
        string $channelDescription = ''
    ): string {
        $base = rtrim($baseUrl, '/');
        $channelTitle = $channelTitle !== '' ? $channelTitle : Seo::siteName() . ' — новости';
        $channelDescription = $channelDescription !== '' ? $channelDescription : 'Новости: услуги, встречи и события.';

        $out = '<?xml version="1.0" encoding="UTF-8"?>' . "\n";
        $out .= '<rss version="2.0" xmlns:atom="http://www.w3.org/2005/Atom">' . "\n";
        $out .= "  <channel>\n";
        $out .= '    <title>' . static::escape($channelTitle) . "</title>\n";
        $out .= '    <link>' . static::escape($base . $prefix) . "</link>\n";
        $out .= '    <description>' . static::escape($channelDescription) . "</description>\n";
        $out .= "    <language>ru</language>\n";
        $out .= '    <lastBuildDate>' . date(DATE_RSS) . "</lastBuildDate>\n";
        $out .= '    <atom:link href="' . static::escape($base . $feedPath) . '" rel="self" type="application/rss+xml"/>' . "\n";

        foreach (static::items($handle) as $record) {
            $link = $base . $prefix . trim((string) $record->slug) . '/';
            $title = trim((string) $record->title);
            $description = trim((string) $record->excerpt);

            if ($description === '') {
                $description = static::plain((string) $record->content);
            }

            $out .= "    <item>\n";
            $out .= '      <title>' . static::escape($title) . "</title>\n";
            $out .= '      <link>' . static::escape($link) . "</link>\n";
            $out .= '      <guid isPermaLink="true">' . static::escape($link) . "</guid>\n";
            $out .= '      <pubDate>' . static::date($record) . "</pubDate>\n";

            if ($description !== '') {
                $out .= '      <description>' . static::escape($description) . "</description>\n";
            }

            $out .= "    </item>\n";
        }

        $out .= "  </channel>\n";
        $out .= '</rss>' . "\n";

        return $out;
    }

    /**
     * items — опубликованные записи раздела, свежие сверху.
     *
     * @return array<int, \Tailor\Models\EntryRecord>
     */
    public static function items(string $handle = self::HANDLE): array
    {
        $records = Content::records($handle);

        usort($records, function ($left, $right) {
            $order = static::timestamp($right) <=> static::timestamp($left);

            return $order !== 0 ? $order : strcmp((string) $left->slug, (string) $right->slug);
        });

        return array_slice($records, 0, static::LIMIT);
    }

    /**
     * date — дата записи в формате RFC 2822.
     */
    protected static function date($record): string
    {
        $timestamp = static::timestamp($record);

        return $timestamp > 0 ? date(DATE_RSS, $timestamp) : date(DATE_RSS);
    }

    /**
     * timestamp — момент публикации записи.
     */
    protected static function timestamp($record): int
    {
        $value = $record->published_at ?: $record->created_at;

        if ($value instanceof \DateTimeInterface) {
            return $value->getTimestamp();
        }

        return $value ? (int) strtotime((string) $value) : 0;
    }

    /**
     * plain — текст без разметки для описания новости.
     */
    protected static function plain(string $html): string
    {
        $spaced = preg_replace('/<[^>]+>/u', ' ', $html) ?? $html;
        $text = trim(preg_replace('/\s+/u', ' ', strip_tags($spaced)) ?? '');

        if (mb_strlen($text) <= 300) {
            return $text;
        }

        return rtrim(mb_substr($text, 0, 300)) . '…';
    }

    /**
     * escape — экранирование для XML.
     */
    protected static function escape(string $value): string
    {
        return htmlspecialchars($value, ENT_QUOTES | ENT_XML1, 'UTF-8');
    }
}

<?php
use Elv\Site\Classes\PageBuilder;
use Elv\Site\Classes\Seo;

function onStart()
{
    $record = PageBuilder::find((string) $this->param('slug'));

    if (!$record) {
        $this->controller->setStatusCode(404);
        $this->page->title = 'Страница не найдена — ' . \Elv\Site\Classes\Brand::get('name');
        $this->page->seo_description = 'Запрошенная страница не найдена.';
    }
    else {
        Seo::applyRecord($this->page, $record, '');
    }

    $prepared = PageBuilder::prepare($record);

    $this['builder'] = $record;
    $this['rows'] = $prepared['rows'];
    $this['toc'] = $prepared['toc'];
}
?>
