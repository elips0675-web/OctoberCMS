#!/usr/bin/env node
/**
 * Проверка 110 адресов сайта на живом October (`docs/qa-checklist.md` §B).
 *
 * Инвентарь берётся из `docs/content.md`: разделы 1.2–1.7 дают слаги
 * детальных страниц, раздел 1.1 — корневые адреса. Скрипт не подменяет
 * DOM-сверку с `preview/`, он отвечает на другой вопрос: адрес отвечает,
 * отдаёт непустую страницу и не тянет внешний CSS.
 *
 * 111 адресов инвентаря = 110 страниц (13 корневых + 6 списков + 91 детальная)
 * плюс 404, который проверяется адресом `/non-existent/`. Корневой `/blog/`
 * добавлен позже источника: блог собран на Tailor, в эталоне его нет.
 *
 *   node tools/check-urls.mjs [--base=http://127.0.0.1:8080]
 *
 * Требуется запущенный `php artisan serve` (порт по умолчанию 8080).
 */

import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const baseArg = process.argv.find((arg) => arg.startsWith('--base='));
const base = (baseArg ? baseArg.slice('--base='.length) : 'http://127.0.0.1:8080').replace(/\/$/, '');

const expect = {
    2: { prefix: 'books', count: 24 },
    3: { prefix: 'printing', count: 37 },
    4: { prefix: 'news', count: 10 },
    5: { prefix: 'services', count: 8 },
    6: { prefix: 'events', count: 6 },
    7: { prefix: 'collections', count: 6 },
};

const rootPages = [
    '/',
    '/books/',
    '/printing/',
    '/news/',
    '/services/',
    '/events/',
    '/collections/',
    '/about/',
    '/auth/',
    '/contacts/',
    '/feedback/',
    '/personal/',
    '/privacy/',
    '/rules/',
    '/search/',
    '/sitemap/',
    '/subscribe/',
    '/terms/',
    '/blog/',
];

function readInventory() {
    const doc = fs.readFileSync(path.join(root, 'docs/content.md'), 'utf8');
    const lines = doc.split(/\r?\n/);
    const urls = [];
    const problems = [];

    let section = null;
    for (const line of lines) {
        const heading = /^### 1\.(\d)\./.exec(line);
        if (heading) {
            section = Number(heading[1]);
            continue;
        }
        if (/^## /.test(line)) {
            section = null;
            continue;
        }
        if (!section || !expect[section]) {
            continue;
        }

        const found = line.match(/`([a-z0-9-]+)`/g) || [];
        for (const token of found) {
            const slug = token.slice(1, -1);
            urls.push('/' + expect[section].prefix + '/' + slug + '/');
        }
    }

    for (const [number, group] of Object.entries(expect)) {
        const actual = urls.filter((url) => url.startsWith('/' + group.prefix + '/')).length;
        if (actual !== group.count) {
            problems.push(
                `раздел 1.${number}: в docs/content.md ${actual} слагов, ожидалось ${group.count}`
            );
        }
    }

    return { urls: [...rootPages, ...urls], problems };
}

const standalone = new Set(['/personal/']);

function h1Count(html) {
    return (html.match(/<h1[\s>]/g) || []).length;
}

function inlineCssBytes(html) {
    const match = /<style data-elv-css="styles">([\s\S]*?)<\/style>/i.exec(html);

    return match ? Buffer.byteLength(match[1], 'utf8') : 0;
}

async function main() {
    const { urls, problems: inventoryProblems } = readInventory();
    const failures = [...inventoryProblems];

    if (urls.length !== 110) {
        failures.push(`адресов ${urls.length}, ожидалось 110 (111-й — 404 по /non-existent/)`);
    }

    let checked = 0;
    let mainDocument = null;

    for (const url of urls) {
        let response;
        try {
            response = await fetch(base + url, { redirect: 'manual' });
        } catch (error) {
            failures.push(`${url}: запрос не выполнен (${error.message})`);
            continue;
        }

        const html = await response.text();
        checked += 1;

        if (response.status !== 200) {
            failures.push(`${url}: HTTP ${response.status}`);
            continue;
        }
        if (html.length < (standalone.has(url) ? 500 : 5000)) {
            failures.push(`${url}: документ ${html.length} знаков — пустая страница`);
            continue;
        }
        if (/Fatal error|Whoops|Exception found/i.test(html)) {
            failures.push(`${url}: страница содержит текст ошибки`);
        }
        if (!standalone.has(url) && !/<main[\s>]/.test(html)) {
            failures.push(`${url}: нет <main>`);
        }
        if (h1Count(html) !== 1) {
            failures.push(`${url}: заголовков h1 — ${h1Count(html)}, ожидался 1`);
        }
        if (html.includes('fonts.googleapis') || html.includes('fonts.gstatic')) {
            failures.push(`${url}: подключены внешние шрифты Google`);
        }
        if (/<link[^>]+rel=["']stylesheet["']/i.test(html)) {
            failures.push(`${url}: внешняя таблица стилей`);
        }
        if (!/lang="ru"/.test(html)) {
            failures.push(`${url}: нет lang="ru"`);
        }

        if (url === '/') {
            mainDocument = html;
        }
    }

    const notFound = await fetch(base + '/non-existent/');
    const notFoundHtml = await notFound.text();
    if (notFound.status !== 404) {
        failures.push(`/non-existent/: HTTP ${notFound.status}, ожидался 404`);
    }
    if (!/Страница не найдена/.test(notFoundHtml)) {
        failures.push('/non-existent/: нет страницы 404, отдана пустота');
    }

    const withQuery = await fetch(base + '/services/cafe/?utm_source=test');
    if (withQuery.status !== 200) {
        failures.push(`/services/cafe/?utm_source=test: HTTP ${withQuery.status}`);
    }

    if (mainDocument) {
        const chars = mainDocument.length;
        if (chars > 190412) {
            failures.push(`главная: ${chars} знаков, бюджет 190 412`);
        }
        const css = inlineCssBytes(mainDocument);
        if (css > 57795) {
            failures.push(`главная: инлайн styles.css ${css} байт, бюджет 57 795`);
        }
    }

    if (failures.length > 0) {
        console.error('FAIL: ' + failures.length + ' проблем');
        for (const line of failures) {
            console.error('  - ' + line);
        }
        process.exit(1);
    }

    console.log(`OK: ${checked} страниц отвечают 200, /non-existent/ — 404 со страницей, внешнего CSS нет`);
    console.log(`главная: ${mainDocument.length} знаков, инлайн styles.css ${inlineCssBytes(mainDocument)} байт`);
}

main().catch((error) => {
    console.error('FAIL: ' + error.message);
    process.exit(1);
});

#!/usr/bin/env node
/**
 * SEO-периметр живого сайта: robots.txt, карта сайта, 301 и базовая микроразметка.
 *
 * Что проверяется (M20):
 *   - `/robots.txt` — 200 `text/plain`, `Disallow: /admin/`, директива `Sitemap`
 *     с адресом этого же сайта;
 *   - `/sitemap.xml` — 200 `application/xml`, `<lastmod>` у каждого адреса,
 *     каждый `<loc>` отвечает 200, закрытые в robots адреса в карту не попали,
 *     и **все** адреса инвентаря `docs/content.md` (кроме закрытых) в карте есть;
 *   - старые адреса 1С-Битрикс отдают 301 на новые (детальная — с сохранением
 *     `utm_*`, без `CODE`);
 *   - на главной есть JSON-LD организации (`Organization` или его подтип,
 *     задаётся настройкой темы) + `WebSite` с `SearchAction`, и это
 *     валидный JSON, а не «почти JSON»;
 *   - на детальной странице узел своего вида (`Article`, `Event`, `Product`,
 *     `Service`), у подборки — только общие узлы, у всех — `og:image`
 *     и `twitter:image` (P2-4, Stage 9);
 *   - `/rss.xml` — 200 `application/rss+xml`, RSS 2.0, ссылка канала на
 *     `/news/`, у каждой записи `title`, абсолютный `link`, `guid` и
 *     разбираемый `pubDate`, число записей совпадает с числом новостей
 *     в карте, на главной есть `<link rel="alternate">` (P2-6, Stage 11).
 *
 *   node tools/check-seo.mjs [--base=http://127.0.0.1:8080]
 *
 * Требуется запущенный `php artisan serve` (порт по умолчанию 8080).
 */

import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const baseArg = process.argv.find((arg) => arg.startsWith('--base='));
const base = (baseArg ? baseArg.slice('--base='.length) : 'http://127.0.0.1:8080').replace(/\/$/, '');

const closed = ['/auth/', '/personal/', '/search/'];

const orgTypes = [

const isOrgType = (type) => orgTypes.includes(type);

const recordTypes = {
    news: 'Article',
    events: 'Event',
    books: 'Product',
    printing: 'Product',
    services: 'Service',
};

const sections = {

    3: { prefix: 'printing', count: 37 },
    4: { prefix: 'news', count: 10 },
    5: { prefix: 'services', count: 8 },
    6: { prefix: 'events', count: 6 },
    7: { prefix: 'collections', count: 6 },
};

const rootPages = [
    '/',

    '/printing/',
    '/news/',
    '/services/',
    '/events/',
    '/collections/',
    '/about/',
    '/auth/',

    '/personal/',
    '/privacy/',
    '/rules/',
    '/search/',
    '/sitemap/',
    '/subscribe/',
    '/terms/',
    '/blog/',

const redirects = [
    { from: '/index.php', to: '/' },
    { from: '/sitemap.php', to: '/sitemap.xml' },
    { from: '/about/index.php', to: '/about/' },
    { from: '/services/index.php', to: '/services/' },
    { from: '/news/detail.php', to: '/news/' },
    { from: '/books/detail.php?CODE=bad%20code', to: '/books/' },

function readInventory() {
    const doc = fs.readFileSync(path.join(root, 'docs/content.md'), 'utf8');
    const urls = [];

    let section = null;
    for (const line of doc.split(/\r?\n/)) {
        const heading = /^### 1\.(\d)\./.exec(line);
        if (heading) {
            section = Number(heading[1]);
            continue;
        }
        if (/^## /.test(line)) {
            section = null;
            continue;

        if (!section || !sections[section]) {
            continue;

        for (const token of line.match(/`([a-z0-9-]+)`/g) || []) {
            urls.push('/' + sections[section].prefix + '/' + token.slice(1, -1) + '/');
        }
    }

    return urls;
}

function detailSlug(url) {
    const match = /^\/([a-z]+)\/([a-z0-9-]+)\/$/.exec(url);

    return match ? { prefix: match[1], slug: match[2] } : null;
}

/**
 * checkBreadcrumbs проверяет `BreadcrumbList` страницы раздела.
 *
 * Крошки собираются из адреса (P2-22): «Главная → раздел → страница».
 * Гейт ловит не только отсутствие узла, но и `position` вне порядка,
 * пустые имена, относительные `item` и последнюю крошку не на своём адресе.
 */
function checkBreadcrumbs(url, graph) {
    const failures = [];
    const node = graph.find((item) => item['@type'] === 'BreadcrumbList');

    if (!node) {
        return [`${url}: нет узла BreadcrumbList`];
    }

    const items = Array.isArray(node.itemListElement) ? node.itemListElement : [];

    if (items.length < 2) {
        return [`${url}: в BreadcrumbList ${items.length} элементов`];
    }

    items.forEach((item, index) => {
        if (item.position !== index + 1) {
            failures.push(`${url}: position ${item.position} на месте ${index + 1}`);
        }

        if (!item.name) {
            failures.push(`${url}: крошка ${index + 1} без имени`);
        }

        if (!/^https?:\/\//.test(item.item || '')) {
            failures.push(`${url}: крошка ${index + 1} без абсолютного адреса`);
        }
    });

    const last = items[items.length - 1];

    if (last.item !== base + url) {
        failures.push(`${url}: последняя крошка ведёт на «${last.item}»`);
    }

    return failures;
}

/**
 * metaContent возвращает content мета-тега или пустую строку.
 */
function metaContent(html, attribute, name) {
    const pattern = new RegExp(`<meta ${attribute}="${name}" content="([^"]*)">`);
    const match = pattern.exec(html);

    return match ? match[1] : '';
}

/**
 * isAbsoluteAssetUrl проверяет, что это абсолютный адрес с путём.
 *
 * Прежняя проверка `https?:\\/\\/[^"]+` пропускала склеенный хост без слэша
 * (`http://hostthemes/elv/...`): для соцсетей такая ссылка — не картинка,
 * а проверка была зелёной.
 */
function isAbsoluteAssetUrl(value) {
    try {
        const parsed = new URL(value);

        return /^https?:$/.test(parsed.protocol) && parsed.pathname !== '/';
    }
    catch {
        return false;
    }
}

/**
 * checkDetailPage проверяет микроразметку и карточку шеринга детальной страницы.
 */
function checkDetailPage(url, html) {
    const failures = [];
    const prefix = url.split('/')[1];
    const expected = recordTypes[prefix];

    const match = /<script type="application\/ld\+json">([\s\S]*?)<\/script>/.exec(html);
    if (!match) {
        return [`${url}: нет <script type="application/ld+json">`];
    }

    let graph = [];
    try {
        const data = JSON.parse(match[1]);
        graph = Array.isArray(data['@graph']) ? data['@graph'] : [];
    }
    catch (error) {
        return [`${url}: JSON-LD не разбирается (${error.message})`];
    }

    const types = graph.map((node) => node['@type']);

    if (expected) {
        const node = graph.find((item) => item['@type'] === expected);

        if (!node) {
            failures.push(`${url}: нет узла ${expected} (есть ${types.join(', ') || 'ничего'})`);
        }
        else if (expected === 'Article' && (!node.headline || !node.datePublished)) {
            failures.push(`${url}: у Article нет headline или datePublished`);
        }
        else if (expected === 'Event' && (!node.startDate || !node.location)) {
            failures.push(`${url}: у Event нет startDate или location`);
        }
        else if (expected === 'Product' && !node.sku) {
            failures.push(`${url}: у Product нет sku`);
        }
        else if (expected === 'Service' && !node.serviceType) {
            failures.push(`${url}: у Service нет serviceType`);
        }
    }
    else {
        if (types.some((type) => !isOrgType(type) && type !== 'WebSite' && type !== 'BreadcrumbList')) {
            failures.push(`${url}: у подборки лишние узлы (${types.join(', ')})`);
        }
    }

    failures.push(...checkBreadcrumbs(url, graph));

    const ogImage = metaContent(html, 'property', 'og:image');
    if (!isAbsoluteAssetUrl(ogImage)) {
        failures.push(`${url}: og:image не абсолютный адрес картинки («${ogImage}»)`);
    }

    const twitterImage = metaContent(html, 'name', 'twitter:image');
    if (!isAbsoluteAssetUrl(twitterImage)) {
        failures.push(`${url}: twitter:image не абсолютный адрес картинки («${twitterImage}»)`);
    }

    const canonical = /<link rel="canonical" href="([^"]+)">/.exec(html);
    if (!canonical) {
        failures.push(`${url}: нет canonical`);
    }
    else if (canonical[1] !== base + url) {
        failures.push(`${url}: canonical «${canonical[1]}», ожидался «${base}${url}»`);
    }

    return failures;
}

async function main() {
    const failures = [];

    const robots = await fetch(base + '/robots.txt');
    const robotsBody = await robots.text();
    if (robots.status !== 200) {
        failures.push(`/robots.txt: HTTP ${robots.status}`);
    }
    if (!/text\/plain/.test(robots.headers.get('content-type') || '')) {
        failures.push(`/robots.txt: тип ${robots.headers.get('content-type')}`);
    }
    if (!robotsBody.includes('Disallow: /admin/')) {
        failures.push('/robots.txt: нет Disallow: /admin/');
    }
    if (!robotsBody.includes('Sitemap: ' + base + '/sitemap.xml')) {
        failures.push(`/robots.txt: директива Sitemap не указывает на ${base}/sitemap.xml`);
    }
    if (!robotsBody.includes('Clean-param:')) {
        failures.push('/robots.txt: нет Clean-param для фильтров');
    }

    const sitemap = await fetch(base + '/sitemap.xml');
    const xml = await sitemap.text();
    if (sitemap.status !== 200) {
        failures.push(`/sitemap.xml: HTTP ${sitemap.status}`);
    }
    if (!/application\/xml/.test(sitemap.headers.get('content-type') || '')) {
        failures.push(`/sitemap.xml: тип ${sitemap.headers.get('content-type')}`);
    }
    if (!xml.startsWith('<?xml version="1.0" encoding="UTF-8"?>')) {
        failures.push('/sitemap.xml: нет XML-пролога');
    }

    const blocks = [...xml.matchAll(/<url>([\s\S]*?)<\/url>/g)].map((match) => match[1]);
    const locs = blocks.map((block) => {
        const loc = /<loc>([^<]+)<\/loc>/.exec(block);
        const lastmod = /<lastmod>([^<]+)<\/lastmod>/.exec(block);

        return {
            loc: loc ? loc[1] : '',
            lastmod: lastmod ? lastmod[1] : '',
        };
    });

    const inventory = [...rootPages, ...readInventory()];
    const expected = inventory.filter((url) => !closed.includes(url));

    if (locs.length !== expected.length) {
        failures.push(`/sitemap.xml: адресов ${locs.length}, ожидалось ${expected.length}`);
    }

    const seen = new Set();

    for (const entry of locs) {
        if (!entry.loc.startsWith(base + '/')) {
            failures.push(`/sitemap.xml: <loc> ${entry.loc} не на этом сайте`);
            continue;
        }
        if (!/^\d{4}-\d{2}-\d{2}$/.test(entry.lastmod)) {
            failures.push(`/sitemap.xml: у ${entry.loc} lastmod «${entry.lastmod}»`);
        }

        const local = entry.loc.slice(base.length) || '/';
        seen.add(local);

        if (closed.includes(local)) {
            failures.push(`/sitemap.xml: закрытый в robots адрес ${local} попал в карту`);
        }
    }

    for (const url of expected) {
        if (!seen.has(url)) {
            failures.push(`/sitemap.xml: адреса ${url} нет в карте`);
        }
    }

    for (const url of seen) {
        const response = await fetch(base + url, { redirect: 'manual' });
        if (response.status !== 200) {
            failures.push(`/sitemap.xml: ${url} отвечает ${response.status}`);
            continue;
        }

        const detail = /^\/([a-z]+)\/[a-z0-9-]+\/$/.exec(url);
        if (detail) {
            failures.push(...checkDetailPage(url, await response.text()));
        }
    }

    const home = await (await fetch(base + '/')).text();
    const ld = /<script type="application\/ld\+json">([\s\S]*?)<\/script>/.exec(home);
    if (!ld) {
        failures.push('главная: нет <script type="application/ld+json">');
    }
    else {
        let data = null;
        try {
            data = JSON.parse(ld[1]);
        }
        catch (error) {
            failures.push('главная: JSON-LD не разбирается (' + error.message + ')');
        }

        if (data) {
            const graph = Array.isArray(data['@graph']) ? data['@graph'] : [];
            const org = graph.find((node) => isOrgType(node['@type']));
            const website = graph.find((node) => node['@type'] === 'WebSite');

            if (!org) {
                failures.push('главная: в JSON-LD нет узла организации (Organization)');
            }
            else {
                if (!org.address || !org.address.streetAddress) {
                    failures.push('главная: в JSON-LD организации нет адреса');
                }
                if (!org.openingHoursSpecification) {
                    failures.push('главная: в JSON-LD организации нет часов работы');
                }
            }

            if (!website || !website.potentialAction) {
                failures.push('главная: в JSON-LD нет WebSite с SearchAction');
            }
        }
    }

    const rss = await fetch(base + '/rss.xml');
    const rssBody = await rss.text();

    if (rss.status !== 200) {
        failures.push(`/rss.xml: HTTP ${rss.status}`);
    }
    if (!/application\/rss\+xml/.test(rss.headers.get('content-type') || '')) {
        failures.push(`/rss.xml: тип ${rss.headers.get('content-type')}`);
    }
    if (!rssBody.startsWith('<?xml version="1.0" encoding="UTF-8"?>')) {
        failures.push('/rss.xml: нет XML-пролога');
    }
    if (!/<rss version="2\.0"/.test(rssBody)) {
        failures.push('/rss.xml: корневой элемент не RSS 2.0');
    }

    const channelLink = /<channel>[\s\S]*?<link>([^<]+)<\/link>/.exec(rssBody);
    if (!channelLink || channelLink[1] !== base + '/news/') {
        failures.push(`/rss.xml: ссылка канала «${channelLink ? channelLink[1] : 'нет'}», ожидалась «${base}/news/»`);
    }

    const rssItems = [...rssBody.matchAll(/<item>([\s\S]*?)<\/item>/g)].map((match) => match[1]);
    const expectedNews = expected.filter((url) => /^\/news\/[a-z0-9-]+\/$/.test(url)).length;
    const expectedItems = Math.min(expectedNews, 20);

    if (rssItems.length !== expectedItems) {
        failures.push(`/rss.xml: записей ${rssItems.length}, ожидалось ${expectedItems} (новостей ${expectedNews})`);
    }

#!/usr/bin/env node
/**
 * Заголовки безопасности живого сайта: CSP, X-Frame-Options, Referrer-Policy,
 * Permissions-Policy, HSTS-условие и nonce инлайнового скрипта.
 *
 * Что проверяется (P1-4, Stage 10):
 *   - каждая публичная страница отдаёт полный набор статических заголовков;
 *   - CSP содержит `default-src 'self'`, `frame-ancestors 'none'`,
 *     `object-src 'none'`, `script-src 'self' 'nonce-…'`, `style-src` с
 *     `'unsafe-inline'` (инлайновый CSS темы) и **не** содержит `unsafe-eval`;
 *   - nonce из заголовка совпадает с атрибутом `nonce` инлайнового скрипта
 *     фильтров (`/news/`, `/books/`, `/events/`), а инлайновых скриптов без
 *     nonce на странице нет;
 *   - два запроса получают разные nonce (значение одноразовое);
 *   - админка (`BACKEND_URI`, по умолчанию `/admin`) остаётся без CSP
 *     и с `X-Frame-Options: SAMEORIGIN` — иначе ломаются виджеты October;
 *   - страница 404 получает те же заголовки, а не «голый» ответ.
 *
 *   node tools/check-headers.mjs [--base=http://127.0.0.1:8080]
 *
 * Требуется запущенный `php artisan serve` (порт по умолчанию 8080).
 */

import fs from 'node:fs';
import path from 'node:path';
import { fileURLToPath } from 'node:url';

const root = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..');
const baseArg = process.argv.find((arg) => arg.startsWith('--base='));
const base = (baseArg ? baseArg.slice('--base='.length) : 'http://127.0.0.1:8080').replace(/\/$/, '');

const pages = [
    '/',
    '/books/',
    '/printing/',
    '/news/',
    '/services/',
    '/events/',
    '/collections/',
    '/about/',
    '/contacts/',
    '/feedback/',
    '/subscribe/',
    '/search/',
];

const inlinePages = ['/news/', '/books/', '/events/'];

function backendUri() {
    const envFile = path.join(root, '.env');

    if (!fs.existsSync(envFile)) {
        return 'admin';
    }

    const line = fs.readFileSync(envFile, 'utf8')
        .split(/\r?\n/)
        .find((row) => /^BACKEND_URI=/.test(row));

    if (!line) {
        return 'admin';
    }

    return line.slice('BACKEND_URI='.length).trim().replace(/^["']|["']$/g, '').replace(/^\/|\/$/g, '') || 'admin';
}

function nonceOf(csp) {
    const match = /'nonce-([^']+)'/.exec(csp || '');

    return match ? match[1] : '';
}

async function main() {
    const failures = [];
    const seenNonces = new Set();

    for (const url of pages) {
        const response = await fetch(base + url);
        const headers = response.headers;

        if (response.status !== 200) {
            failures.push(`${url}: HTTP ${response.status}`);
            continue;
        }

        const expected = {
            'x-content-type-options': 'nosniff',
            'x-frame-options': 'DENY',
            'referrer-policy': 'strict-origin-when-cross-origin',
        };

        for (const [name, value] of Object.entries(expected)) {
            const actual = headers.get(name);
            if (actual !== value) {
                failures.push(`${url}: ${name} «${actual || 'нет'}», ожидался «${value}»`);
            }
        }

        const permissions = headers.get('permissions-policy') || '';
        for (const feature of ['camera=()', 'microphone=()', 'geolocation=()']) {
            if (!permissions.includes(feature)) {
                failures.push(`${url}: Permissions-Policy без ${feature}`);
            }
        }

        const csp = headers.get('content-security-policy');

        if (!csp) {
            failures.push(`${url}: нет Content-Security-Policy`);
            continue;
        }

        const required = [
            "default-src 'self'",
            "base-uri 'self'",
            "object-src 'none'",
            "frame-ancestors 'none'",
            "form-action 'self'",
            "img-src 'self' data:",
            "style-src 'self' 'unsafe-inline'",
            "connect-src 'self'",
        ];

        for (const directive of required) {
            if (!csp.includes(directive)) {
                failures.push(`${url}: CSP без «${directive}»`);
            }
        }

        if (csp.includes('unsafe-eval')) {
            failures.push(`${url}: CSP разрешает unsafe-eval`);
        }

        const nonce = nonceOf(csp);

        if (!nonce) {
            failures.push(`${url}: CSP без nonce для инлайновых скриптов`);
            continue;
        }

        seenNonces.add(nonce);

        if (!csp.includes(`script-src 'self' 'nonce-${nonce}'`)) {
            failures.push(`${url}: script-src не содержит nonce этого ответа`);
        }

        if (headers.get('strict-transport-security')) {
            failures.push(`${url}: HSTS на http-ответе`);
        }

        if (!inlinePages.includes(url)) {
            continue;
        }

        const html = await response.text();
        const inline = /<script nonce="([^"]*)">/.exec(html);

        if (!inline) {
            failures.push(`${url}: инлайновый скрипт фильтров без nonce`);
        }
        else if (inline[1] !== nonce) {
            failures.push(`${url}: nonce скрипта «${inline[1]}» не совпадает с заголовком «${nonce}»`);
        }

        const rogue = /<script(?![^>]*\bsrc=)(?![^>]*\bnonce=)(?![^>]*type="application\/ld\+json")[^>]*>/i.exec(html);
        if (rogue) {
            failures.push(`${url}: инлайновый скрипт без nonce — ${rogue[0].slice(0, 80)}`);
        }
    }

    if (seenNonces.size !== pages.length) {
        failures.push(`nonce повторяется: ${seenNonces.size} разных значений на ${pages.length} страниц`);
    }

    const admin = `/${backendUri()}`;
    const backend = await fetch(base + admin);
    const backendHeaders = backend.headers;

    if (![200, 302].includes(backend.status)) {
        failures.push(`${admin}: HTTP ${backend.status}, ожидался 200 или 302`);
    }
    if (backendHeaders.get('content-security-policy')) {
        failures.push(`${admin}: CSP на админке ломает виджеты October`);
    }
    if (backendHeaders.get('x-frame-options') !== 'SAMEORIGIN') {
        failures.push(`${admin}: X-Frame-Options «${backendHeaders.get('x-frame-options') || 'нет'}», ожидался SAMEORIGIN`);
    }
    if (backendHeaders.get('x-content-type-options') !== 'nosniff') {
        failures.push(`${admin}: нет X-Content-Type-Options`);
    }

    const missing = await fetch(base + '/non-existent-page-check/');
    if (missing.status !== 404) {
        failures.push(`/non-existent-page-check/: HTTP ${missing.status}, ожидался 404`);
    }
    if (missing.headers.get('x-frame-options') !== 'DENY') {
        failures.push('404: нет X-Frame-Options DENY');
    }
    if (!missing.headers.get('content-security-policy')) {
        failures.push('404: нет Content-Security-Policy');
    }

    if (failures.length > 0) {
        console.error('FAIL: ' + failures.length + ' проблем');
        for (const line of failures) {
            console.error('  - ' + line);
        }
        process.exit(1);
    }

    console.log(`OK: заголовки на ${pages.length} страницах, CSP с уникальным nonce, админка без CSP (${admin}), 404 с заголовками`);
}

main().catch((error) => {
    console.error('FAIL: ' + error.message);
    process.exit(1);
});

const CONCURRENCY = 6;

/**
 * fetchText забирает страницу и возвращает тело либо null с записью ошибки.
 */
async function fetchText(url) {

    if (!response.ok) {
        failures.push(`${url.slice(base.length) || '/'}: HTTP ${response.status}`);

/**
 * sitemapPaths вытаскивает пути из `<loc>` карты сайта, отбрасывая чужие хосты.
 */
function sitemapPaths(xml) {
    return [...xml.matchAll(/<loc>([^<]+)<\/loc>/g)]
        .map((match) => {
            let url;

            try {
                url = new URL(match[1]);
            }
            catch {
                return null;
            }

            return url.origin === origin ? url.pathname : null;
        })
        .filter((pathname) => pathname !== null);
}

/**
 * internalLinks вытаскивает внутренние пути из `href` страницы.
 */
function internalLinks(html, currentPath) {
    const links = new Set();

    for (const match of html.matchAll(/href="([^"]+)"/g)) {
        const raw = match[1];

        if (raw.startsWith('mailto:') || raw.startsWith('tel:') || raw.startsWith('javascript:')) {
            continue;
        }

        let url;

        try {
            url = new URL(raw, base + currentPath);
        }
        catch {
            continue;
        }

        if (url.origin !== origin) {
            continue;
        }

        const pathname = url.pathname;

        if (pathname === currentPath) {
            continue;
        }

        links.add(pathname);
    }

    return links;
}

/**
 * orphansOf возвращает адреса без входящих ссылок (главная и исключения — нет).
 */
function orphansOf(paths, incoming) {
    return paths.filter((pathname) => {
        if (pathname === '/' || ALLOWED_ORPHANS.includes(pathname)) {
            return false;
        }

        return (incoming.get(pathname) || 0) === 0;
    });
}

/**
 * mapLimit выполняет fn по списку с ограничением параллельности.
 */
async function mapLimit(items, limit, fn) {
    const results = new Array(items.length);
    let index = 0;

    const workers = Array.from({ length: Math.min(limit, items.length) }, async () => {
        while (index < items.length) {
            const current = index++;
            results[current] = await fn(items[current], current);
        }
    });

    await Promise.all(workers);

    return results;
}

function selfTest() {
    const problems = [];

    const html = [
        '<a href="/a/">внутренняя</a>',
        '<a href="/a/?page=2#top">с параметром и якорем</a>',
        '<a href="/a/">дубль</a>',

        '<a href="/c/d/">вложенная</a>',
    ].join('');

    const links = internalLinks(html, '/a/');

    if (!links.has('/a/') && links.size === 0) {
        problems.push('internalLinks не нашёл ни одной ссылки');
    }

    if (!links.has('/c/d/')) {
        problems.push('internalLinks потерял вложенный путь /c/d/');
    }

    if (links.has('/b/')) {
        problems.push('internalLinks пропустил внешний хост');
    }

    const selfLinks = internalLinks('<a href="/a/">сама</a>', '/a/');

    if (selfLinks.has('/a/')) {
        problems.push('ссылка на себя считается входящей');
    }

    const paths = sitemapPaths('<urlset><url><loc>' + base + '/a/</loc></url><url><loc>https://other.example/x/</loc></url></urlset>');

    if (paths.length !== 1 || paths[0] !== '/a/') {
        problems.push('sitemapPaths: ' + JSON.stringify(paths));
    }

    const incoming = new Map([['/', 5], ['/a/', 1], ['/b/', 0], ['/404', 0]]);
    const orphans = orphansOf(['/', '/a/', '/b/', '/404'], incoming);

    if (JSON.stringify(orphans) !== JSON.stringify(['/b/'])) {
        problems.push('orphansOf: ' + JSON.stringify(orphans));
    }

    if (problems.length > 0) {
        console.error('FAIL: ' + problems.length + ' проблем самопроверки');
        for (const line of problems) {
            console.error('  - ' + line);
        }
        process.exit(1);
    }

    console.log('OK: самопроверка разбора ссылок и поиска сирот');
}

        throw new Error('в карте сайта нет адресов');

        console.error('FAIL: ' + failures.length + ' проблем');
        for (const line of failures) {
            console.error('  - ' + line);

}

main().catch((error) => {

#!/usr/bin/env node
/**
 * Контракт публичного JSON (`/api/v1/...`, P2-14) на живом сайте.
 *
 * Что проверяется:
 *   - каждый раздел из `Sitemap::SECTIONS` отвечает 200 `application/json`,
 *     `meta.section` совпадает с handle, `meta.count` — с длиной `data`;
 *   - у каждой записи есть обязательные поля контракта (`id`, `slug`, `title`,
 *     `excerpt`, `image`, `url`, `updated_at`), `url` абсолютный;
 *   - **каждый `url` из списка открывается 200**: API не отдаёт запись,
 *     которой нет в вёрстке (фильтр `applyVisibleFrontend` не разошёлся
 *     с темой);
 *   - `{handle}/{slug}` первой записи отвечает 200 с тем же `id`;
 *   - неизвестный handle — 404 в JSON, `limit=0` — 422.
 *
 *   node tools/check-api.mjs [--base=http://127.0.0.1:8080]
 *
 * Требуется запущенный `php artisan serve` (порт по умолчанию 8080).
 */

const baseArg = process.argv.find((arg) => arg.startsWith('--base='));
const base = (baseArg ? baseArg.slice('--base='.length) : 'http://127.0.0.1:8080').replace(/\/$/, '');

const SECTIONS = ['books', 'printing', 'news', 'services', 'events', 'collections'];

const REQUIRED = ['id', 'slug', 'title', 'excerpt', 'image', 'url', 'updated_at'];

const failures = [];

/**
 * getJson забирает JSON и возвращает пару «ответ, тело» либо null с ошибкой.
 */
async function getJson(path) {
    const response = await fetch(base + path);
    const text = await response.text();

    try {
        return { response, data: JSON.parse(text) };
    }
    catch {
        failures.push(`${path}: тело не JSON (${text.slice(0, 80)})`);
        return { response, data: null };
    }
}

function checkItem(path, item) {
    for (const field of REQUIRED) {
        if (!Object.prototype.hasOwnProperty.call(item, field)) {
            failures.push(`${path}: в записи нет поля ${field}`);
        }
    }

    if (!/^https?:\/\//.test(item.url || '')) {
        failures.push(`${path}: url не абсолютный (${item.url})`);
    }
}

async function main() {
    let total = 0;

    for (const section of SECTIONS) {
        const path = `/api/v1/${section}`;
        const { response, data } = await getJson(path);

        if (response.status !== 200) {
            failures.push(`${path}: HTTP ${response.status}`);
            continue;
        }

        if (!(response.headers.get('content-type') || '').includes('application/json')) {
            failures.push(`${path}: content-type ${response.headers.get('content-type')}`);
        }

        if (!data || !data.meta || data.meta.section !== section) {
            failures.push(`${path}: meta.section = ${data && data.meta ? data.meta.section : 'нет'}`);
            continue;
        }

        if (!Array.isArray(data.data) || data.meta.count !== data.data.length) {
            failures.push(`${path}: meta.count не совпадает с длиной data`);
            continue;
        }

        for (const item of data.data) {
            checkItem(path, item);
        }

        total += data.data.length;

        const first = data.data[0];

        if (!first) {
            continue;
        }

        const detail = await getJson(`${path}/${first.slug}`);

        if (detail.response.status !== 200) {
            failures.push(`${path}/${first.slug}: HTTP ${detail.response.status}`);
        }
        else if (!detail.data || !detail.data.data || detail.data.data.id !== first.id) {
            failures.push(`${path}/${first.slug}: id не совпал со списком`);
        }

        for (const item of data.data) {
            const page = await fetch(base + new URL(item.url).pathname, { redirect: 'manual' });

            if (page.status !== 200) {
                failures.push(`${new URL(item.url).pathname}: страница отдала ${page.status}`);
            }
        }
    }

    const unknown = await getJson('/api/v1/unknown');

    if (unknown.response.status !== 404) {
        failures.push(`/api/v1/unknown: HTTP ${unknown.response.status}, ожидался 404`);
    }

    const badLimit = await getJson('/api/v1/books?limit=0');

    if (badLimit.response.status !== 422) {
        failures.push(`/api/v1/books?limit=0: HTTP ${badLimit.response.status}, ожидался 422`);
    }

    if (failures.length > 0) {
        console.error('FAIL: ' + failures.length + ' проблем');
        for (const line of failures) {
            console.error('  - ' + line);
        }
        process.exit(1);
    }

    console.log(`OK: API ${SECTIONS.length} разделов, ${total} записей, поля контракта, страницы записей 200, 404/422 на ошибки`);
}

main().catch((error) => {
    console.error('FAIL: ' + error.message);
    process.exit(1);
});

 * export-static — статическая демо-копия шаблона для Vercel.
 *
 * Vercel не исполняет PHP, поэтому October CMS там отдаётся файлом: команда
 * снимает готовый HTML с локального сервера, переписывает базовый адрес,

 * Запуск: php tools/export-static.php --origin=https://octobercms.vercel.app
 * Нужен поднятый сервер: php artisan serve --port=8080 --host=127.0.0.1
 */

$options = getopt('', ['base::', 'origin::', 'out::', 'stable!', 'manifest::']);

$stable = ($options['stable'] ?? true) !== false;

$base = rtrim($options['base'] ?? 'http://127.0.0.1:8080', '/');
$origin = rtrim($options['origin'] ?? 'https://octobercms.vercel.app', '/');

function staticFetch(string $url): array
{
    $handle = curl_init($url);
    curl_setopt_array($handle, [
        CURLOPT_RETURNTRANSFER => true,
        CURLOPT_FOLLOWLOCATION => true,
        CURLOPT_MAXREDIRS => 3,
        CURLOPT_TIMEOUT => 30,
        CURLOPT_USERAGENT => 'ElvThemeStaticExport/1.0',
    ]);

    $body = curl_exec($handle);
    $status = (int) curl_getinfo($handle, CURLINFO_RESPONSE_CODE);
    $error = curl_error($handle);
    curl_close($handle);

    if ($body === false) {
        fwrite(STDERR, 'fetch failed: ' . $url . ' — ' . $error . PHP_EOL);

        return [0, ''];
    }

    return [$status, (string) $body];
}

function staticRemove(string $directory): void
{
    if (!is_dir($directory)) {
        return;
    }

    $items = new RecursiveIteratorIterator(
        new RecursiveDirectoryIterator($directory, FilesystemIterator::SKIP_DOTS),
        RecursiveIteratorIterator::CHILD_FIRST
    );

    foreach ($items as $item) {
        $item->isDir() ? rmdir($item->getPathname()) : unlink($item->getPathname());
    }

    rmdir($directory);
}

function staticCopy(string $source, string $target): void
{
    if (!is_dir($source)) {
        return;
    }

    if (!is_dir($target) && !mkdir($target, 0775, true) && !is_dir($target)) {
        throw new RuntimeException('cannot create ' . $target);
    }

    $items = new RecursiveIteratorIterator(
        new RecursiveDirectoryIterator($source, FilesystemIterator::SKIP_DOTS),
        RecursiveIteratorIterator::SELF_FIRST
    );

    foreach ($items as $item) {
        $destination = $target . '/' . $items->getSubPathname();

        if ($item->isDir()) {
            if (!is_dir($destination)) {
                mkdir($destination, 0775, true);
            }

            continue;
        }

        copy($item->getPathname(), $destination);
    }
}

function staticPagePath(string $out, string $path): string
{
    return $path === '/'
        ? $out . '/index.html'
        : ($path === '/404' ? $out . '/404.html' : $out . '/' . trim($path, '/') . '/index.html');
}

function staticFinalize(string $html, string $base, string $origin, string $inject): string
{
    $html = str_replace($base, $origin, $html);
    $quoted = preg_quote($origin, '~');
    $html = (string) preg_replace('~(\b(?:href|src|action|poster|content)=")' . $quoted . '/~i', '$1/', $html);
    $html = (string) preg_replace('~(\b(?:href|src|action|poster)=")' . $quoted . '"~i', '$1/"', $html);
    $html = (string) preg_replace('~(<link rel="canonical" href=")/~i', '$1' . $origin . '/', $html);
    $html = (string) preg_replace('~(<meta property="og:url" content=")/~i', '$1' . $origin . '/', $html);

    $bar = '<div id="elv-demo-bar" role="status">Статическая демо-версия шаблона ELV на October CMS: формы, брони и заказы отключены. Рабочая версия — на PHP-хостинге.</div>';

    if (!str_contains($html, 'elv-demo-bar')) {
        $html = (string) preg_replace('~(<body[^>]*>)~i', '$1' . $bar, $html, 1);
    }

    return str_replace('</body>', $inject . '</body>', $html);
}

function staticStable(string $html): string
{
    $html = (string) preg_replace('~\bnonce="[^"]*"~i', 'nonce="static-demo"', $html);
    $html = (string) preg_replace('~(<meta name="csrf-token" content=")[^"]*(")~i', '$1static-demo$2', $html);
    $html = (string) preg_replace('~(<input type="hidden" name="_token" value=")[^"]*(")~i', '$1static-demo$2', $html);

    return $html;
}

function staticWrite(string $out, string $path, string $html, bool $stable = true): void
{
    $file = staticPagePath($out, $path);

    if (!is_dir(dirname($file))) {
        mkdir(dirname($file), 0775, true);
    }

    file_put_contents($file, $stable ? staticStable($html) : $html);
}

 *
 * Тема вкладывает `bootstrap` и `styles` в каждую страницу (по ~110 КБ),
 * из-за чего статическая копия из 113 страниц весит 16 МБ. Если содержимое
 * блока совпадает на всех страницах, оно пишется один раз в
 * `themes/elv/assets/css/`, а `<style>` заменяется на `<link>`.

function staticInlineCss(string $out): void

    $pages = [];

    $iterator = new RecursiveIteratorIterator(
        new RecursiveDirectoryIterator($out, FilesystemIterator::SKIP_DOTS)
    );

    foreach ($iterator as $file) {
        if ($file->isFile() && $file->getExtension() === 'html') {
            $pages[] = $file->getPathname();
        }
    }

    $variants = [];
    $content = [];

    foreach ($pages as $page) {
        $html = (string) file_get_contents($page);
        preg_match_all('~<style data-elv-css="([a-z0-9-]+)"[^>]*>([\s\S]*?)</style>~i', $html, $matches, PREG_SET_ORDER);

        foreach ($matches as $match) {
            $variants[$match[1]][md5($match[2])] = true;
            $content[$match[1]][md5($match[2])] = $match[2];
        }
    }

    $target = $out . '/themes/elv/assets/css';

    if (!is_dir($target) && !mkdir($target, 0775, true) && !is_dir($target)) {
        throw new RuntimeException('cannot create ' . $target);
    }

    foreach ($variants as $name => $hashes) {

                $html

            file_put_contents($page, $html);

# Публичное демо на Vercel — что нажать

Демо шаблона — статический снимок `demo-static/` (110 страниц, формы отключены),
его отдаёт Vercel по `vercel.json:2` (`outputDirectory: demo-static`).
Проект уже создан: `https://vercel.com/vadims-projects-74cb21aa/octobercms`,
репозиторий подключён — `https://github.com/elips0675-web/OctoberCMS`.

## Почему просит войти через Google

Проверено 19.11.2026 снаружи, без авторизации:

```text
https://octobercms-git-main-vadims-projects-74cb21aa.vercel.app
  → 302 Location: https://vercel.com/sso-api?url=…&nonce=…
  → Set-Cookie: _vercel_sso_nonce=…

https://octobercms.vercel.app
  → 404 X-Vercel-Error: DEPLOYMENT_NOT_FOUND

https://octobercms-vadims-projects-74cb21aa.vercel.app
  → 404 X-Vercel-Error: DEPLOYMENT_NOT_FOUND
```

Два независимых факта:

1. **Deployment Protection = Vercel Authentication** (это и есть «вход гугла»).
   Редирект на `sso-api` — его подпись; страница защищена целиком, поэтому её
   видит только владелец аккаунта Vercel.
2. **Продакшен-деплоя нет**: домен проекта `*.vercel.app` отвечает
   `DEPLOYMENT_NOT_FOUND`, значит деплой не повышен до Production — живой
   только алиас ветки `git-main`.

Правки в репозитории этого не лечат: защита и повышение деплоя живут
в настройках проекта Vercel, а не в `vercel.json` (такой опции в схеме нет).

## Два действия владельца (≈2 минуты)

1. **Снять защиту.** Проект `octobercms` → **Settings** → **Deployment
   Protection** → **Vercel Authentication**: выбрать **Disabled** → **Save**.
   Заодно проверить, что не включены `Password Protection` и `Trusted IPs`.
2. **Повысить деплой до продакшена.** **Deployments** → у свежей записи со
   статусом **Ready** меню **⋯** → **Promote to Production**.
   После этого появляется домен `https://octobercms-vadims-projects-74cb21aa.vercel.app`.

Если в проекте нет ни одного успешного деплоя — **Deployments** → **Redeploy**
у последнего коммита `main` (сборка идёт по `vercel.json`, Framework Preset
**Other**, Build Command пустой).

1. Пересобрать снимок под домен демо (canonical, sitemap, RSS):

   ```powershell

   node tools/check-static-demo.mjs --origin=https://octobercms-vadims-projects-74cb21aa.vercel.app
   ```

2. В `README.md` (раздел «Демо») уже записано, что публичного адреса нет;
   поле **Website** репозитория на GitHub переключено на
   `https://octobercms.vercel.app/` (19.11.2026, было — витрина

**Settings** → **Domains** → **Add** (например `demo.elv.studio`), в DNS

выпускает сам. После привязки — та же пересборка с `--origin=https://demo.elv.studio`.


Без токена скрипт выходит с кодом 2 и ничего не меняет; токен не печатается и
не сохраняется. Код возврата `0` — демо открылось (`200`, без
`x-vercel-error`).

## Ручные запросы (если нужен контроль по шагам)

Если у владельца есть токен (`vercel.com/account/tokens`), защиту и продакшен
можно переключить без интерфейса:

```powershell
$token = "<токен>"
$project = "octobercms"

# 1. снять Vercel Authentication (ssoProtection = null)
curl.exe -sS -X PATCH "https://api.vercel.com/v9/projects/$project" `
  -H "Authorization: Bearer $token" -H "Content-Type: application/json" `
  -d "{\"ssoProtection\":null,\"passwordProtection\":null}"

# 2. узнать последний деплой и повысить его
curl.exe -sS "https://api.vercel.com/v6/deployments?projectId=$project&limit=1" `
  -H "Authorization: Bearer $token"
curl.exe -sS -X POST "https://api.vercel.com/v10/projects/$project/promote/<deploymentId>" `
  -H "Authorization: Bearer $token"
```

Токен в репозиторий и в переписку не попадает: он нужен только для этих двух
запросов и удаляется из переменной сессии после.

# Архитектура

## 1. Место в стеке

Сайт — обычное Laravel-приложение October CMS. Поток запроса:

```
nginx/apache → index.php (корень проекта) → October bootstrap → plugin routes / CMS page
  → компонент (Header, Menu, Breadcrumbs, List, Detail, Form, Search)
  → Eloquent / Tailor-запись
  → Blade-шаблон темы elv → HTML (инлайн CSS)
```

Никакого SPA, клиентского роутинга и сборки фронтенда. JS — `bootstrap.bundle` +
`script.js` как в источнике.

## 2. Плагин `elv/site`

## 3. Tailor или Eloquent: правило разделения

- **Tailor** — read-mostly контент, который правится в админке полями: `news`, `events`,
  `collections`, `books`, `services`, `printing`.
- **Eloquent** — всё, что требует транзакций, уникальных индексов, статусных переходов,
  идемпотентности и тестов: `requests`, `bookings`, `orders`, `slots`, `subscribers`,
  `push_subscriptions`.

Исключение: `services` меняется и в Tailor (контент), и в Eloquent-таблице `slots`
(расписание). Источник правды по расписанию — Eloquent, blueprint услуги хранит только
презентационные поля.

### 6.2. Служебные

| Маршрут | Назначение | Контракт |
|---|---|---|
| `POST /feedback/` | обратная связь (P0-4) | `elv_requests` + письмо в очередь; `201` JSON / `302 ?sent=1#form`; `422` на пустые поля |

| `POST /api/bookings` — **план (P1-16)** | REST для Telegram-бота | `Bearer`-токен, идемпотентность по `external_id`, `200/201/400/401/409/422/429`; маршрута пока нет |
| `POST /ajax/push/subscribe`, `POST /ajax/push/unsubscribe` — **план (M9)** | Web Push | VAPID из `.env`; маршрутов пока нет |
| `GET /api/v1/{handle}` | JSON контента для приложения и бота (P2-14) | `200` `{data, meta:{section,count}}`, `404` JSON для чужого раздела, `422` для `limit` вне 1…100 |
| `GET /api/v1/{handle}/{slug}` | одна запись | `200` `{data}`, `404` для отключённой и несуществующей |

REST API контента (`classes/Http/ApiController.php`) отдаёт поля
`id, slug, title, excerpt, image, url, updated_at` (константа `ApiController::FIELDS`
— это и есть контракт). `handle` — префикс адреса из `Sitemap::SECTIONS`
(`books`, `printing`, `news`, `services`, `events`, `collections`), поэтому
API и карта сайта не могут разойтись: список разделов один. Выборка идёт через
`Content::records()`/`Content::record()`, то есть с тем же фильтром
`applyVisibleFrontend()`: отключённая запись не попадает ни в список, ни в
`{slug}`. `url` и `image` абсолютные (картинка строится через
`Media::themeUrl()`), `excerpt` — из анонса или из текста без разметки (до 200
символов), `limit` по умолчанию не ограничен, потолок —
`ApiController::MAX_LIMIT` (100). Группа без `web`: сессия и CSRF API не нужны,
лимит запросов — `throttle:60,1`, ответы кешируются на минуту
(`Cache-Control: public, max-age=60`).


Формы (`plugins/elv/site/routes.php`) сидят в группе `web`: обязателен
`_token` (в шаблоне — `{{ elv_csrf()|raw }}`), лимит `throttle:30,1`.
Поля источника `PROPERTY[EMAIL][0]` разбирает `classes/FormFields.php` в плоские
`EMAIL`, `DATE`, … Возврат — на страницу формы из `Referer` того же хоста,
иначе на `fallbackPath()`: чужой `Referer` не превращается в открытый редирект.

`GET /sitemap.xml` и `GET /robots.txt` — `classes/Http/SeoController.php`:
файлы генерируются, а не лежат в `public/`, потому что домен появится только на
M17, а `Sitemap:`/`<loc>` с чужим доменом уводят поисковиков на чужой сайт.
Адрес берётся из запроса (`url('/')`), как в источнике (`sitemap.php`).

### 6.3. Админка

| Раздел (меню «Библиотека») | Контроллер (`plugins/elv/site/controllers`) | Право |
|---|---|---|
| Обращения | `Requests.php` — `elv/site/requests` | `elv.site.requests` |
| Брони | `Bookings.php` — `elv/site/bookings` | `elv.site.bookings` |
| Заказы | `Orders.php` — `elv/site/orders` | `elv.site.orders` |
| Подписчики | `Subscribers.php` — `elv/site/subscribers` | `elv.site.subscribers` |

Права и меню объявлены в `Plugin.php` (`registerPermissions()`,
`registerNavigation()`): у каждого пункта свой `permissions`, поэтому оператор
видит только выданные разделы. Списки — `ListController` + `columns.yaml`
(поиск по теме, e-mail, телефону, услуге, товару, адресу); правка —
`FormController` + `fields.yaml`, где данные гостя только для чтения
(`disabled`), а редактируется один `status`.

Смена статуса — единственное действие админки, и оно не обходит бизнес-логику:

- `Request::saving` / `Order::saving` ставят `date_processed`, когда статус стал
  «Выполнено»/«Отменён»;
- `Booking::updating` при статусе «Отменена» освобождает слот через
  `BookingService::releaseSlot()` (общий код с `BookingService::cancel()`), а
  возврат отменённой брони в работу запрещён `ApplicationException` — иначе
  слот вернулся бы в продажу без брони.

Демо-подсказка «часть слотов занята» и массовые действия (удаление, экспорт)
не делались: их нет в приёмке.

Карта сайта собирается из тех же записей, что показывает тема: отключённая или

`Sitemap::PAGES` роняет `tests/SeoTest.php`, а не выпадает из индекса молча.

(`classes/Http/RequestPath.php:18`): Symfony принимает `/index.php` за
`SCRIPT_NAME` и `path()` отдаёт `/`, иначе старый адрес главной не отличить от
самой главной. `CODE` отбрасывается, остальные параметры (`utm_*`, `PAGEN_1`)
переносятся в новый адрес. Middleware подключён в `Plugin.php:117` только вне
консоли — юнит-тесты гоняют его напрямую; рядом `ManagedRedirect`
(`Plugin.php:125`, идёт после), `SecurityHeaders` (`Plugin.php:121`) и
`CompressResponse` (`Plugin.php:113`).

Мета записи: у шести blueprints (`news`, `events`, `books`, `services`,
`printing`, `collections`) есть `seo_title` и `seo_description`,
`Seo::applyRecord()` (`classes/Seo.php:169`) выставляет их в `<head>`, пустые
поля откатываются на имя записи и `excerpt` (обрезка 160 знаков). Canonical
собирается как `getSchemeAndHttpHost() . $path` (`Seo::canonical()`,
`classes/Seo.php:84`), иначе `url('/x/')` срезает хвостовой слэш и адрес
расходится с `<loc>` карты. Микроразметка записи (`Article`, `Event`,
`Product`, `Service`) — `Seo::recordNode()` (`classes/Seo.php:399`), Stage 9.
Панель в записи и разбор контента — Stage 33, редиректы редактора — Stage 34
(таблица выше). Остаток: сводка по всем записям (P2-35), CWV после домена
(M17); FAQ отклонён — на сайте нет видимого блока «вопрос-ответ».

## 7. Очереди и планировщик

| Задача | Тип | Что делает |
|---|---|---|
| `FeedbackNotification` | `Mailable` в очереди | письмо сотрудникам об обращении с сайта |
| `BookingNotification` | `Mailable` в очереди | письмо сотрудникам о новой брони |
| `BookingConfirmation` | `Mailable` в очереди | подтверждение брони клиенту |
| `OrderNotification` | `Mailable` в очереди | письмо сотрудникам о заказе сувенира |
| `SubscriptionConfirmation` | `Mailable` в очереди | double opt-in и ссылка отписки |
| `NotifyStaffPush` | план | Web Push персоналу о брони |
| `elv:health-report` | schedule (ежечасно) | JSON-строка состояния: ответ БД, очередь, диск, компоненты |
| `elv:slots-generate` | schedule (ежедневно, 03:10) | генерация слотов из расписания услуг |
| `elv:search-index` | план (M8) | переиндексация BM25 |
| `ExpireUnconfirmedBookings` | план | снятие просроченных броней |

Отдельных job-классов нет: письмо ставит в очередь контроллер формы
(`Mail::queue(new …)`), письма — `plugins/elv/site/classes/Mail/`. Очередь
`QUEUE_CONNECTION=database`, поэтому без воркера (`php artisan queue:work`)
письма лежат в таблице `jobs` и не уходят — это требование хостинга, а не код.

SMTP у владельца нет: `MAIL_MAILER=log`, письма проверяются в
`storage/logs/system.log` и Mailpit. Код обязан работать с реальным SMTP без правок.

## 8. Медиа и CSS

- Картинки: варианты 480/960/1280 в WebP/AVIF собирает `tools/build-image-variants.php`
  (GD, `imagewebp`, `imageavif`) — переносится как есть.
- `<picture>` выводит `Picture`-компонент: `srcset`, `width`/`height`,
  `fetchpriority="high"` у hero, `loading="lazy"` ниже первого экрана.
- CSS: `bootstrap.min.css` (51 758 байт) собирается `tools/purge-css.mjs` из
  `bootstrap.source.min.css` (232 803 байта) по классам, которые реально есть в
  `themes/elv/**` и `plugins/elv/**`; каталог `controllers` в обход сканера —
  это разметка админки, её стилизует тема backend, а не CSS сайта. Вместе со
  `styles.css` (84 995 байт) результат минифицируется хелпером
  `Media::minifyCss()` и встраивается инлайном в layout. Кеш — по `mtime`,
  отдельного `styles.min.css` не существует (рассинхрон «файл есть, а его никто
  не пересобрал» недопустим).

## 9. Зависимости

Плагин не тянет сторонних пакетов, кроме идущих с October. Запрещено: демо-плагины,
RainLab.Blog, «универсальные» SEO-пакеты, Tailwind, React, Vite, Google Fonts.
Причина: лишние запросы и лишний CSS убивают 100/100/100/100.

Порядок установки сторонних плагинов и список допустимых — `docs/plugins.md`.
Там же зафиксировано, почему не берём Winter CMS: нет Tailor и Laravel 9 без
security-фиксов.
