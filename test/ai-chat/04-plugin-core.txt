<?php namespace Elv\Site\Classes\Http;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

/**
 * SecurityHeaders — CSP и защитные заголовки для публичных страниц.
 *
 * Источник (`elv_security.php`, 77 строк) отдавал CSP, X-Frame-Options,
 * Referrer-Policy и Permissions-Policy. В October заголовки обязан ставить
 * периметр (nginx), но shared-хостинг и `php artisan serve` остаются без него,
 * поэтому логика дублируется middleware'ом: nginx скрывает эти же заголовки
 * от PHP и добавляет свои (`nginx.conf`), а CSP живёт только здесь — в ней
 * одноразовый nonce, который nginx знать не может.
 *
 * Админка исключена из CSP: October подключает инлайновые скрипты и eval
 * в бэкенде, и строгая политика ломает редакторы и виджеты. Для неё остаются
 * статические заголовки с X-Frame-Options SAMEORIGIN (всплывающие окна).
 */
class SecurityHeaders
{
    protected const NONCE_ATTRIBUTE = 'elv_csp_nonce';

    protected static ?string $nonce = null;

    public function handle(Request $request, Closure $next): Response
    {
        self::$nonce = base64_encode(random_bytes(16));
        $request->attributes->set(self::NONCE_ATTRIBUTE, self::$nonce);

        $response = $next($request);

        if (!$response instanceof Response) {
            return $response;
        }

        return $this->apply($request, $response);
    }

    /**
     * nonce — значение для инлайновых скриптов страницы.
     */
    public static function nonce(): string
    {
        return self::$nonce ?? '';
    }

    /**
     * reset — сбрасывает nonce между запросами в тестах.
     */
    public static function reset(): void
    {
        self::$nonce = null;
    }

    protected function apply(Request $request, Response $response): Response
    {
        $headers = $response->headers;
        $backend = $this->isBackend($request);

        $headers->set('X-Content-Type-Options', 'nosniff');
        $headers->set('Referrer-Policy', 'strict-origin-when-cross-origin');
        $headers->set('Permissions-Policy', 'camera=(), microphone=(), geolocation=()');
        $headers->set('X-Frame-Options', $backend ? 'SAMEORIGIN' : 'DENY');

        if ($request->isSecure()) {
            $headers->set('Strict-Transport-Security', 'max-age=31536000; includeSubDomains');
        }

        if (!$backend) {
            $headers->set('Content-Security-Policy', $this->policy($request));
        }

        return $response;
    }

    protected function isBackend(Request $request): bool
    {
        $uri = trim((string) config('backend.uri', 'admin'), '/');

        if ($uri === '') {
            return false;
        }

        return $request->is($uri) || $request->is($uri . '/*');
    }

    protected function policy(Request $request): string
    {
        $directives = [
            "default-src 'self'",
            "base-uri 'self'",
            "object-src 'none'",
            "frame-ancestors 'none'",
            "form-action 'self'",
            "img-src 'self' data:",
            "font-src 'self'",
            "style-src 'self' 'unsafe-inline'",
            "script-src 'self' 'nonce-" . self::nonce() . "'",
            "connect-src 'self'",
            "media-src 'self'",
            "manifest-src 'self'",
            "worker-src 'self'",
        ];

        if ($request->isSecure()) {
            $directives[] = 'upgrade-insecure-requests';
        }

        return implode('; ', $directives);
    }
}

<?php namespace Elv\Site\Classes\Http;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

/**
 * CompressResponse — gzip для текстовых ответов и кеш-заголовки для ассетов.
 *
 * Нужен там, где перед PHP нет nginx с gzip: `php artisan serve`, shared-хостинг
 * без настроенного сжатия. С nginx middleware безвреден: уже сжатый ответ он
 * не трогает (сравнивает Accept-Encoding и наличие Content-Encoding).
 */
class CompressResponse
{
    protected const COMPRESSIBLE = [
        'text/html',
        'text/css',
        'text/plain',
        'text/xml',
        'text/javascript',
        'application/javascript',
        'application/json',
        'application/xml',
        'application/rss+xml',
        'application/ld+json',
        'image/svg+xml',
    ];

    protected const MIN_LENGTH = 1024;

    protected const COMPRESSION_LEVEL = 6;

    public function handle(Request $request, Closure $next): Response
    {
        $response = $next($request);

        if (!$this->isCompressible($request, $response)) {
            return $this->withAssetsCache($request, $response);
        }

        $content = $response->getContent();
        $compressed = gzencode($content, self::COMPRESSION_LEVEL);

        if ($compressed === false || strlen($compressed) >= strlen($content)) {
            return $this->withAssetsCache($request, $response);
        }

        $response->setContent($compressed);
        $response->headers->set('Content-Encoding', 'gzip');
        $response->headers->set('Content-Length', (string) strlen($compressed));
        $response->headers->set('Vary', 'Accept-Encoding', false);

        return $this->withAssetsCache($request, $response);
    }

    protected function isCompressible(Request $request, Response $response): bool
    {
        if (!$response instanceof Response || $response instanceof \Symfony\Component\HttpFoundation\BinaryFileResponse) {
            return false;
        }

        if ($response->headers->has('Content-Encoding')) {
            return false;
        }

        if (!$this->clientAcceptsGzip($request->headers->get('Accept-Encoding', ''))) {
            return false;
        }

        $content = $response->getContent();
        if (!is_string($content) || strlen($content) < self::MIN_LENGTH) {
            return false;
        }

        $contentType = (string) $response->headers->get('Content-Type', '');

        foreach (self::COMPRESSIBLE as $type) {
            if (stripos($contentType, $type) !== false) {
                return true;
            }
        }

        return false;
    }

    protected function clientAcceptsGzip(string $header): bool
    {
        if (stripos($header, 'gzip') === false) {
            return false;
        }

        if (preg_match('/gzip\s*;\s*q=([0-9.]+)/i', $header, $matches)) {
            return (float) $matches[1] > 0;
        }

        return true;
    }

    protected function withAssetsCache(Request $request, Response $response): Response
    {
        if ($response->headers->has('Cache-Control')) {
            return $response;
        }

        $path = '/' . ltrim($request->path(), '/');
        $maxAge = $this->assetMaxAge($path);

        if ($maxAge === null) {
            return $response;
        }

        $response->headers->set('Cache-Control', 'public, max-age=' . $maxAge . ', immutable');

        return $response;
    }

    protected function assetMaxAge(string $path): ?int
    {
        if (preg_match('#^/(themes|plugins|modules)/[^/]+/(assets|resources)/#', $path)) {
            return 31536000;
        }

        if (str_starts_with($path, '/storage/app/')) {
            return 604800;
        }

        return null;
    }
}

<?php namespace Elv\Site\Classes;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Queue;

/**
 * Health — проверка живости БД, кэша и очереди.
 *
 * Отвечает только «да/нет» по компонентам и числа для отчёта. Текст исключения
 * наружу не отдаётся: он попадает в лог, а в ответе — имя компонента, чтобы
 * health-страница не стала источником утечки (DSN, SQL, пароль).
 */
class Health
{
    /**
     * check — состояние трёх компонентов и общий вердикт.
     *
     * @return array{ok: bool, checks: array<string, bool>}
     */
    public function check(): array
    {
        $checks = [
            'db' => $this->database(),
            'cache' => $this->cache(),
            'queue' => $this->queue(),
        ];

        return [
            'ok' => !in_array(false, $checks, true),
            'checks' => $checks,
        ];
    }

    /**
     * metrics — числа для часового отчёта.
     *
     * @return array<string, mixed>
     */
    public function metrics(): array
    {
        return [
            'db_ms' => $this->databaseMilliseconds(),
            'queue_size' => $this->queueSize(),
            'disk_free_bytes' => $this->diskFreeBytes(),
            'checks' => $this->check()['checks'],
        ];
    }

    /**
     * database — соединение отвечает на простой запрос.
     */
    protected function database(): bool
    {
        try {
            DB::connection()->select('SELECT 1');

            return true;
        } catch (\Throwable $e) {
            $this->log('db', $e);

            return false;
        }
    }

    /**
     * cache — значение записывается и читается обратно.
     */
    protected function cache(): bool
    {
        try {
            $key = 'elv:health:' . bin2hex(random_bytes(6));
            Cache::put($key, 'ok', 5);

            return Cache::pull($key) === 'ok';
        } catch (\Throwable $e) {
            $this->log('cache', $e);

            return false;
        }
    }

    /**
     * queue — соединение очереди доступно.
     *
     * `sync` проверять нечего: задания выполняются в том же процессе.
     */
    protected function queue(): bool
    {
        try {
            if (config('queue.default') === 'sync') {
                return true;
            }

            Queue::connection()->size();

            return true;
        } catch (\Throwable $e) {
            $this->log('queue', $e);

            return false;
        }
    }

    /**
     * databaseMilliseconds — время ответа БД или null, если она не ответила.
     */
    protected function databaseMilliseconds(): ?float
    {
        $start = microtime(true);

        try {
            DB::connection()->select('SELECT 1');

            return round((microtime(true) - $start) * 1000, 1);
        } catch (\Throwable $e) {
            $this->log('db', $e);

            return null;
        }
    }

    /**
     * queueSize — размер очереди или null, если она недоступна.
     */
    protected function queueSize(): ?int
    {
        try {
            if (config('queue.default') === 'sync') {
                return 0;
            }

            return (int) Queue::connection()->size();
        } catch (\Throwable $e) {
            $this->log('queue', $e);

            return null;
        }
    }

    /**
     * diskFreeBytes — свободное место на диске приложения.
     */
    protected function diskFreeBytes(): ?int
    {
        $free = @disk_free_space(base_path());

        return $free === false ? null : (int) $free;
    }

    /**
     * log — причина отказа уходит в лог, а не в ответ.
     */
    protected function log(string $component, \Throwable $e): void
    {
        \Log::warning('health: ' . $component . ' недоступен', [
            'component' => $component,
            'error' => $e->getMessage(),
        ]);
    }
}

<?php namespace Elv\Site\Classes\Http;

use Illuminate\Http\JsonResponse;
use Elv\Site\Classes\Health;

/**
 * HealthController — ответ `/health` для внешней пробы.
 *
 * Живой сайт — 200, отказ любого компонента — 503. В теле только имена
 * компонентов и «да/нет»: без DSN, SQL и текста исключений.
 */
class HealthController
{
    /**
     * index отдаёт состояние БД, кэша и очереди.
     */
    public function index(Health $health): JsonResponse
    {
        $result = $health->check();

        return response()->json([
            'status' => $result['ok'] ? 'ok' : 'fail',
            'checks' => $result['checks'],
            'time' => now()->toIso8601String(),
        ], $result['ok'] ? 200 : 503, [
            'Cache-Control' => 'no-store',
        ]);
    }
}
